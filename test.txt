cd ~/camera/hik_camera_ws_v2
unzip -o ~/下载/demo_window_update.zip         # 换成压缩包的实际路径
cd hik_camera_ws
source /opt/ros/jazzy/setup.bash
colcon build --symlink-install --packages-select hik_camera_driver

ros2 run hik_camera_driver demo_window.sh open     # 打开：没加载就加载；按过 q 就重新打开
ros2 run hik_camera_driver demo_window.sh close    # 关闭窗口，组件保留（再 open 很快）
ros2 run hik_camera_driver demo_window.sh unload   # 彻底卸载（正式采集前推荐）
ros2 run hik_camera_driver demo_window.sh status   # 查看当前加载了几个
